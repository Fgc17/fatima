## How plugin models work

Inside `fatima.json`, the `model` key maps variables to built-in or plugin-provided model names:

```json
{
  "plugins": ["./fatima.plugin.ts"],
  "model": {
    "ADMIN_PASSWORD": { "type": "strongPassword", "min": 12 }
  }
}
```

Plugins can provide the runtime parser and generator-specific wrappers for those models.

```ts
import type { FatimaPlugin } from "fatima/plugin";

const plugin: FatimaPlugin = {
  models: {
    strongPassword: {
      validate(value) {
        if (!value || value.length < 12) return undefined;
        return value;
      },
      generators: {
        typescript: { type: "string", wrap: "$1" },
        javascript: { type: "string", wrap: "$1" },
        python: { type: "str", wrap: "$1" }
      }
    }
  }
};

export default plugin;
```
