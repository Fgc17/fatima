## Importing the loader

```ts title="env.config.ts"
import { config, providers } from "fatima";

export default config({
  providers: {
    development: [
      providers.local(".env"),
      providers.infisical({
        // This provider can now read auth values loaded by the previous provider
      }),
    ],
  },
});
```
