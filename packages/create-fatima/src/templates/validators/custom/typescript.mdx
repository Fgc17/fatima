```ts
import type { StandardSchemaV1 } from "fatima";

export default config({
  validate: {
    "~standard": {
      version: 1,
      vendor: "custom",
      async validate(env) {
        if (!env.NODE_ENV) {
          return {
            issues: [
              {
                message: "NODE_ENV is required",
                path: ["NODE_ENV"],
              },
            ],
          } satisfies StandardSchemaV1.FailureResult;
        }

        return { value: env };
      },
    },
  },
});
```
