```js
module.exports = config({
  validate: {
    "~standard": {
      version: 1,
      vendor: "custom",
      async validate(env) {
        if (!env.NODE_ENV) {
          return {
            issues: [{ message: "NODE_ENV is required", path: ["NODE_ENV"] }],
          };
        }

        return { value: env };
      },
    },
  },
});
```
